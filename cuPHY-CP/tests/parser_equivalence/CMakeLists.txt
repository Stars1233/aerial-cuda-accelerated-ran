# ---------------------------------------------------------------------------
# Parser equivalence tests + benchmarks.
#
# Per-channel acceptance tests that compare the legacy FAPI populator against
# the new scf_5g_fapi parser (ULSlotProcessor), driven by real FAPI messages
# generated by testmac_core. Both the functional test and the benchmark are
# data-gated at runtime (skip when launch-pattern assets are absent).
#
# Requires the compiled scf_5g_fapi (legacy populator + new parser) and the
# testmac_core library (real FAPI UL_TTI generation).
# ---------------------------------------------------------------------------

if(NOT (ENABLE_TESTS
        AND ENABLE_FAPI_STORE_REPLAY
        AND SCF_FAPI_10_04
        AND TARGET scf_5g_fapi
        AND TARGET testmac_core
        AND TARGET nvphy))
    return()
endif()

# Defensive: the parent cuPHY-CP/tests/CMakeLists.txt already requires these
# before add_subdirectory(parser_equivalence). find_package is idempotent/cached,
# so re-requesting them keeps this subdirectory self-contained if it is ever
# configured via a different include path.
find_package(GTest REQUIRED)
find_package(benchmark REQUIRED)

# Shared include paths: testMAC facade, scf_5g_fapi parser, nvPHY public headers.
set(_parser_equivalence_includes
    ${CMAKE_CURRENT_SOURCE_DIR}/../../testMAC/testMAC
    ${CMAKE_CURRENT_SOURCE_DIR}/../../scfl2adapter/lib/scf_5g_fapi
    ${CMAKE_CURRENT_SOURCE_DIR}/../../cuphyl2adapter/lib/nvPHY)

# Executable lives at build/cuPHY-CP/tests/parser_equivalence/ -> 4 dirs to root
# (fallback only; get_full_path_file() prefers the cuBB_SDK env var).
set(_parser_equivalence_root_relative 4)

# fapi_message_source.cpp is the only TU that includes the testMAC headers; it
# is compiled into each front-end and bridges to the parser-side TUs through
# the raw-FAPI-buffer boundary in fapi_message_source.hpp (the testMAC and
# scf_5g_fapi slot-command headers declare clashing names and cannot share a TU).
set(_parser_equivalence_message_source fapi_message_source.cpp)

# --- Functional equivalence test (GTest) ---
add_executable(test_parser_equivalence
    pusch_parser_equivalence_test.cpp
    ${_parser_equivalence_message_source})
target_include_directories(test_parser_equivalence PRIVATE ${_parser_equivalence_includes})
target_compile_definitions(test_parser_equivalence
    PRIVATE CONFIG_CUBB_ROOT_DIR_RELATIVE_NUM=${_parser_equivalence_root_relative})
target_link_libraries(test_parser_equivalence PRIVATE
    testmac_core
    # Break the scf_5g_fapi <-> nvphy static-archive cycle regardless of link order
    # (same fix as test_cuphycp): nvphy's inlined parsers call back into scf_5g_fapi.
    # nvphy also exposes cuphydriver/include (the nv_phy_module.hpp chain).
    "$<LINK_GROUP:RESCAN,scf_5g_fapi,nvphy>"
    slot_command
    nvlog
    CUDA::cuda_driver  # harness calls the CUDA driver API directly (cuInit / primary context)
    # New target: inherit C++20 + active sanitizer flags. sdk::warnings is
    # intentionally omitted (same as test_scfl2_csirs_slot_helpers): this TU
    # transitively includes scf_5g_fapi.h, whose GNU zero-length arrays trip
    # -Werror=pedantic under the project warning set.
    sdk::options)
target_link_system_libraries(test_parser_equivalence PRIVATE GTest::gtest_main)

include(GoogleTest)
# TEST_PREFIX so the per-case CTest names are filterable by the target name,
# i.e. `ctest -R test_parser_equivalence` (mirrors the test_scfl2adapter usage).
gtest_discover_tests(test_parser_equivalence
    TEST_PREFIX test_parser_equivalence.
    PROPERTIES LABELS cuphy-cp)

# --- Parser benchmark (Google Benchmark) ---
# Build-only target; run manually (e.g. ./bench_parser_equivalence). Skips with
# an error at runtime when the launch-pattern data assets are unavailable.
add_executable(bench_parser_equivalence
    pusch_parser_benchmark.cpp
    ${_parser_equivalence_message_source})
target_include_directories(bench_parser_equivalence PRIVATE ${_parser_equivalence_includes})
target_compile_definitions(bench_parser_equivalence
    PRIVATE CONFIG_CUBB_ROOT_DIR_RELATIVE_NUM=${_parser_equivalence_root_relative})
target_link_libraries(bench_parser_equivalence PRIVATE
    testmac_core
    # Break the scf_5g_fapi <-> nvphy static-archive cycle regardless of link order
    # (same fix as test_cuphycp): nvphy's inlined parsers call back into scf_5g_fapi.
    # nvphy also exposes cuphydriver/include (the nv_phy_module.hpp chain).
    "$<LINK_GROUP:RESCAN,scf_5g_fapi,nvphy>"
    slot_command
    nvlog
    CUDA::cuda_driver  # harness calls the CUDA driver API directly (cuInit / primary context)
    # New target: inherit C++20 + active sanitizer flags. sdk::warnings is
    # intentionally omitted (same as test_scfl2_csirs_slot_helpers): this TU
    # transitively includes scf_5g_fapi.h, whose GNU zero-length arrays trip
    # -Werror=pedantic under the project warning set.
    sdk::options)
target_link_system_libraries(bench_parser_equivalence PRIVATE benchmark::benchmark)

# scf_5g_fapi_metrics.hpp pulls aerial_metrics.hpp under -DAERIAL_METRICS; link the
# target so its include dir is on the path (mirrors test_cuphycp).
if(AERIAL_METRICS)
    target_link_libraries(test_parser_equivalence PRIVATE aerial_metrics)
    target_link_libraries(bench_parser_equivalence PRIVATE aerial_metrics)
endif()
